cmake_minimum_required(VERSION 3.16)
project(anyinfer_hc_post LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(ANYINFER_REFERENCE_ONLY "Build the host-only arithmetic and shape tests" OFF)
include(CTest)
if(BUILD_TESTING OR ANYINFER_REFERENCE_ONLY)
    add_executable(anyinfer-hc-post-reference-test reference_test.cpp)
    add_test(NAME hc-post-reference COMMAND anyinfer-hc-post-reference-test)
endif()
if(ANYINFER_REFERENCE_ONLY)
    return()
endif()

set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit installation")
if(NOT ASCEND_CANN_PACKAGE_PATH)
    set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/ascend-toolkit/latest)
endif()
if(NOT SOC_VERSION)
    message(FATAL_ERROR "Pass -DSOC_VERSION=<exact SoC reported by aclrtGetSocName>")
endif()
set(RUN_MODE npu)
find_file(ASCENDC_MODULE ascendc.cmake PATHS
    "${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake"
    "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake"
    "${ASCEND_CANN_PACKAGE_PATH}/ascendc_devkit/tikcpp/samples/cmake"
    "${ASCEND_CANN_PACKAGE_PATH}/${CMAKE_SYSTEM_PROCESSOR}-linux/asc/ascendc_kernel_cmake"
    "${ASCEND_CANN_PACKAGE_PATH}/${CMAKE_SYSTEM_PROCESSOR}-linux/tikcpp/ascendc_kernel_cmake"
    NO_DEFAULT_PATH)
if(NOT ASCENDC_MODULE)
    message(FATAL_ERROR "Missing ascendc.cmake in ${ASCEND_CANN_PACKAGE_PATH}. Source the CANN toolkit set_env.sh")
endif()
include("${ASCENDC_MODULE}")
ascendc_library(anyinfer_hc_post_kernel SHARED "${CMAKE_CURRENT_SOURCE_DIR}/hc_post_kernel.cpp")
find_library(ASCENDCL ascendcl PATHS "${ASCEND_CANN_PACKAGE_PATH}/lib64"
    "${ASCEND_CANN_PACKAGE_PATH}/lib64/stub" NO_DEFAULT_PATH)
if(NOT ASCENDCL)
    message(FATAL_ERROR "Cannot find CANN libascendcl")
endif()
add_library(anyinfer_hc_post SHARED hc_post_launch.cpp)
target_link_libraries(anyinfer_hc_post PRIVATE anyinfer_hc_post_kernel "${ASCENDCL}")
target_include_directories(anyinfer_hc_post PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
set_target_properties(anyinfer_hc_post PROPERTIES INSTALL_RPATH "$ORIGIN")
add_executable(anyinfer-hc-post-bench bench.cpp)
target_include_directories(anyinfer-hc-post-bench PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_libraries(anyinfer-hc-post-bench PRIVATE anyinfer_hc_post "${ASCENDCL}")
target_compile_definitions(anyinfer-hc-post-bench PRIVATE ANYINFER_SOC="${SOC_VERSION}")
set_target_properties(anyinfer-hc-post-bench PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS anyinfer_hc_post_kernel anyinfer_hc_post anyinfer-hc-post-bench
    LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES hc_post.h DESTINATION include)
