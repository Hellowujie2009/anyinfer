cmake_minimum_required(VERSION 3.16)
project(anyinfer_head_runner LANGUAGES CXX)
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN development installation")
if(NOT ASCEND_CANN_PACKAGE_PATH)
    set(ASCEND_CANN_PACKAGE_PATH /usr/local/Ascend/ascend-toolkit/latest)
endif()
find_path(ACL_INCLUDE acl/acl.h PATHS "${ASCEND_CANN_PACKAGE_PATH}/include" NO_DEFAULT_PATH)
find_library(ASCENDCL ascendcl PATHS "${ASCEND_CANN_PACKAGE_PATH}/lib64"
             "${ASCEND_CANN_PACKAGE_PATH}/lib64/stub" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT ASCENDCL)
    message(FATAL_ERROR "Missing CANN ACL headers or libascendcl in ${ASCEND_CANN_PACKAGE_PATH}")
endif()
add_executable(anyinfer-head-rank rank.cpp)
target_compile_features(anyinfer-head-rank PRIVATE cxx_std_17)
target_include_directories(anyinfer-head-rank PRIVATE "${ACL_INCLUDE}")
target_link_libraries(anyinfer-head-rank PRIVATE "${ASCENDCL}")
# Newer CANN splits model APIs into a separate library. Earlier releases export them in ascendcl.
find_library(ACL_MDL acl_mdl PATHS "${ASCEND_CANN_PACKAGE_PATH}/lib64"
             "${ASCEND_CANN_PACKAGE_PATH}/lib64/stub" NO_DEFAULT_PATH)
if(ACL_MDL)
    target_link_libraries(anyinfer-head-rank PRIVATE "${ACL_MDL}")
endif()
install(TARGETS anyinfer-head-rank RUNTIME DESTINATION bin)
